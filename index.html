📖 【100%確実に書き込める】Apps Script コード（コピーして貼り付け）
  1. スプレッドシートのメニュー「拡張機能」➔「Apps Script」を開きます。
  2. エディタの内容をすべて消去し、以下のスクリプトを貼り付けて「保存（💾アイコン）」します：
function recordToSheet(params) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["日時", "クラス", "出席番号", "氏名", "挑戦レベル", "分野", "問題ID", "問題タイトル", "文字数", "評価ランク", "生徒の論述解答"]);
    sheet.getRange("A1:K1").setFontWeight("bold").setBackground("#e2e8f0");
  }
  sheet.appendRow([
    params.timestamp || new Date().toLocaleString("ja-JP"),
    params.className || "",
    params.number || "",
    params.name || "",
    params.level || "",
    params.field || "",
    params.problemId || "",
    params.problemTitle || "",
    params.charCount || 0,
    params.rank || "",
    params.answer || ""
  ]);
}

// GETリクエスト対応（最も確実・CORS制限ゼロ）
function doGet(e) {
  try {
    var p = (e && e.parameter) ? e.parameter : {};
    recordToSheet(p);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// POSTリクエスト対応
function doPost(e) {
  try {
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try { data = JSON.parse(e.postData.contents); } catch (err) { data = e.parameter || {}; }
    } else if (e && e.parameter) {
      data = e.parameter;
    }
    recordToSheet(data);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
